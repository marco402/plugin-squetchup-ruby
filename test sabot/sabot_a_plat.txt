
# Fonction magique pour remettre à plat un groupe dessiné de biais
def self.remettre_groupe_a_plat(groupe)
  return unless groupe

  # 1. Trouver la face principale du panneau (la plus grande surface)
  face_principale = groupe.entities.grep(Sketchup::Face).max_by(&:area)
  return unless face_principale

  # 2. Obtenir le vecteur normal de cette face
  normale = face_principale.normal

  # L'objectif est d'aligner cette normale avec l'axe vertical global Z (0,0,1)
  axe_cible = Geom::Vector3d.new(0, 0, 1)

  # Si la face est déjà parfaitement à plat, on ne fait rien
  return if normale.parallel?(axe_cible)

  # 3. Calculer l'axe de rotation (perpendiculaire aux deux vecteurs) et l'angle
  axe_rotation = normale.cross(axe_cible)
  angle_rad = normale.angle_between(axe_cible)

  # 4. Créer la transformation de rotation (autour de l'origine interne du groupe)
  point_pivot = Geom::Point3d.new(0, 0, 0)
  tr_redressement = Geom::Transformation.rotation(point_pivot, axe_rotation, angle_rad)

  # 5. On applique la transformation DIRECTEMENT sur la géométrie interne du groupe 
  # pour corriger le dessin d'origine "de biais" !
  groupe.entities.transform_entities(tr_redressement, groupe.entities.to_a)
end

def find_group_by_id(id,type)
  model = Sketchup.active_model

  model.entities.grep(Sketchup::Group).each do |g|
    return g if g.get_attribute(type, "id") == id
  end

  model.definitions.each do |d|
    d.entities.grep(Sketchup::Group).each do |g|
      return g if g.get_attribute(type, "id") == id
    end
  end

  nil
end
# Fonction pour mesurer la vraie dimension d'une face à plat le long des axes X et Y
def self.mesurer_dimensions_reelles(groupe)
  return [0.mm, 0.mm] unless groupe
  
  # On récupère toutes les arêtes de la face principale
  face_principale = groupe.entities.grep(Sketchup::Face).max_by(&:area)
  return [0.mm, 0.mm] unless face_principale
  
  # On extrait toutes les coordonnées X et Y des sommets (vertices) de cette face
  points = face_principale.vertices.map { |v| v.position }
  x_coords = points.map { |pt| pt.x }
  y_coords = points.map { |pt| pt.y }
  
  # La vraie largeur et profondeur sont la différence entre le maximum et le minimum
  largeur_reelle = x_coords.max - x_coords.min
  profondeur_reelle = y_coords.max - y_coords.min
  
  [largeur_reelle, profondeur_reelle]
end
